    href === "/" ? pathname === "/" : pathname.startsWith(href);

  const TrendIcon =
    mrrTrend > 0 ? TrendingUp : mrrTrend < 0 ? TrendingDown : Minus;

  // ─── Alert Center ──────────────────────────────────────────────
  const [alerts, setAlerts] = useState<SavedAlert[]>([]);
  const [alertPanelOpen, setAlertPanelOpen] = useState(false);
  const [justTriggered, setJustTriggered] = useState(false);
  const alertPanelRef = useRef<HTMLDivElement>(null);
  const triggered = getTriggeredAlerts(alerts);
  const unreadCount = getUnreadTriggeredAlerts(alerts).length;

  const refreshFromStorage = () => setAlerts(loadAlerts());

  const runCheck = async () => {
    const current = loadAlerts();
    if (current.length === 0) return;
    const { updated, newlyTriggered } = await runCheckAllAlerts(current);
    setAlerts(updated);
    if (newlyTriggered.length > 0) {
      setJustTriggered(true);
      setAlertPanelOpen(true);
      setTimeout(() => setJustTriggered(false), 1600);
      setTimeout(() => setAlertPanelOpen(false), 2500);
    }
  };

  useEffect(() => {
    refreshFromStorage(); // initial load
    runCheck(); // check on mount (page load / nav)

    const onUpdate = () => refreshFromStorage();
    window.addEventListener(ALERTS_UPDATED_EVENT, onUpdate); // instant, same-tab (upload-hook)
    window.addEventListener("storage", onUpdate); // cross-tab

    const poll = setInterval(runCheck, 60000); // 60s catch-all

    return () => {
      window.removeEventListener(ALERTS_UPDATED_EVENT, onUpdate);
      window.removeEventListener("storage", onUpdate);
      clearInterval(poll);
    };
  }, []);

  useEffect(() => {
    const onClickOutside = (e: MouseEvent) => {
      if (
        alertPanelOpen &&
        alertPanelRef.current &&
        !alertPanelRef.current.contains(e.target as Node)
      ) {
        setAlertPanelOpen(false);
      }
    };
    document.addEventListener("mousedown", onClickOutside);
    return () => document.removeEventListener("mousedown", onClickOutside);
  }, [alertPanelOpen]);
